cmake_minimum_required(VERSION 3.24)
project(HellbladeTimedBlock VERSION 1.3.5.5 DESCRIPTION "Hellblade Player Time Independence: optional block visual notification" LANGUAGES CXX RC)
set(PROJECT_FRIENDLY_NAME "Hellblade - Player Time Independence")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(REDSAND_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../.build" CACHE PATH "Prepared fixed CommonLib and vcpkg workspace")
set(COMMONLIB "${REDSAND_BUILD}/CommonLibSSE-NG")
set(DEPS "${REDSAND_BUILD}/Precision/build-release/vcpkg_installed/x64-windows-static-md")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS}")
find_package(spdlog CONFIG REQUIRED)
find_package(directxtk CONFIG REQUIRED)
file(GLOB_RECURSE PLUGIN_SOURCES CONFIGURE_DEPENDS "source/src/*.cpp")
configure_file(source/cmake/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
add_library(HellbladeTimedBlock SHARED ${PLUGIN_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
set_source_files_properties(source/src/time_vfx_d3d.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
target_include_directories(HellbladeTimedBlock PRIVATE source/include source/extern "${COMMONLIB}/include" "${COMMONLIB}/extern/openvr/headers" "${DEPS}/include")
target_compile_definitions(HellbladeTimedBlock PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _UNICODE UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 SKSE_SUPPORT_XBYAK=1 SKSE_SUPPORT_PATCH_SAFETY=1 ENABLE_SKYRIM_SE=1 ENABLE_SKYRIM_AE=1 HAS_SKYRIM_MULTI_TARGETING=1 COMMONLIB_VERSION="7.0.0")
target_compile_options(HellbladeTimedBlock PRIVATE /EHsc /utf-8 /permissive- /Zc:preprocessor /Zi)
target_precompile_headers(HellbladeTimedBlock PRIVATE source/include/pch.hpp)
target_link_libraries(HellbladeTimedBlock PRIVATE "${REDSAND_BUILD}/Precision/build-release/src/CommonLibSSE/Release/CommonLibSSE.lib" spdlog::spdlog Microsoft::DirectXTK advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 version shell32 user32)
target_link_options(HellbladeTimedBlock PRIVATE /DEBUG:FULL /INCREMENTAL:NO /OPT:REF /OPT:ICF)
enable_testing()
add_executable(block_time_tests tests/block_time_tests.cpp)
target_include_directories(block_time_tests PRIVATE source/include)
add_test(NAME world_slow_player_normal_curve COMMAND block_time_tests)
add_executable(time_runtime_tests tests/time_runtime_tests.cpp source/src/block_time.cpp)
target_include_directories(time_runtime_tests PRIVATE tests)
target_compile_definitions(time_runtime_tests PRIVATE REDSAND_TIME_TEST)
add_test(NAME world_player_separation_and_foreign_restore COMMAND time_runtime_tests)
add_executable(attack_time_tests tests/attack_time_tests.cpp source/src/block_time.cpp)
target_include_directories(attack_time_tests PRIVATE tests source/include)
target_compile_definitions(attack_time_tests PRIVATE REDSAND_TIME_TEST)
add_test(NAME attack_probability_and_session_priority COMMAND attack_time_tests)
add_executable(lightning_geometry_tests tests/lightning_geometry_tests.cpp)
target_include_directories(lightning_geometry_tests PRIVATE source/include)
add_test(NAME procedural_lightning_geometry_and_budget COMMAND lightning_geometry_tests)
add_executable(lightning_d3d_tests tests/lightning_d3d_tests.cpp source/src/time_vfx_d3d.cpp)
target_include_directories(lightning_d3d_tests PRIVATE source/include)
target_compile_definitions(lightning_d3d_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(lightning_d3d_tests PRIVATE d3d11 d3dcompiler dxgi)
add_test(NAME actual_d3d11_render_state_and_fade COMMAND lightning_d3d_tests)
